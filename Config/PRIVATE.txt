# ==============================================================
# 放在 fork 后仓库的 Config/PRIVATE.txt
# Scripts/Settings.sh 会在 make defconfig 之前把它"最后"追加到 .config，
# 因此这里的 =y / =n 优先级高于 Config/AIROHA-ONU.txt 和 Config/GENERAL.txt。
#
# 本文件按"你现有那台 477 包机器的实际用法"定制：
#   去掉不用的 → 补齐缺少的 → 锁定关键项
# 每个包名都对照 TEST 产物的 config 校验过，确认符号真实存在。
# ==============================================================


# ==============================================================
# 一、排除：科学上网（SSR 全家桶）
# 上游 Config/GENERAL.txt 里是 =y
# ==============================================================
CONFIG_PACKAGE_luci-app-ssr-plus=n
CONFIG_PACKAGE_xray-core=n
CONFIG_PACKAGE_dns2tcp=n

# mihomo 会编进固件，但它的面板 luci-app-nikki 在这套源码树里根本不存在
# （Packages.sh 里 nikki 的 UPDATE_PACKAGE 行是注释掉的），编了没人调用，白占约 20MB
CONFIG_PACKAGE_mihomo=n
CONFIG_PACKAGE_luci-app-nikki=n


# ==============================================================
# 二、排除：DDNS
# 这套 CI 默认就是关的（你现在机器上有是 VSEAN 固件带的），这里锁死
# ==============================================================
CONFIG_PACKAGE_luci-app-ddns=n
CONFIG_PACKAGE_luci-app-ddns-go=n
CONFIG_PACKAGE_ddns-go=n
CONFIG_PACKAGE_ddns-scripts=n


# ==============================================================
# 三、排除：Watchcat（同上，默认未开，锁死）
# ==============================================================
CONFIG_PACKAGE_luci-app-watchcat=n
CONFIG_PACKAGE_watchcat=n


# ==============================================================
# 四、排除：iStore 应用商店（GENERAL.txt 里是 luci-app-store=y）

CONFIG_PACKAGE_luci-app-store=n

# 五、保留：Lucky（上游已是 =y，写明是为了让你一眼看到）
# ==============================================================
CONFIG_PACKAGE_luci-app-lucky=y
CONFIG_PACKAGE_lucky=y
CONFIG_PACKAGE_luci-i18n-lucky-zh-cn=y


# ==============================================================
# 六、补齐：WireGuard
# 上游只开了 kmod-wireguard，缺 wg 命令和 LuCI 协议支持，
# 等于"模块在、界面没有、配不了"。你现在的 wg0（10.10.10.1/24，端口 12321）必须靠这三条。
# ==============================================================
CONFIG_PACKAGE_kmod-wireguard=y
CONFIG_PACKAGE_wireguard-tools=y
CONFIG_PACKAGE_luci-proto-wireguard=y
CONFIG_PACKAGE_qrencode=y


# ==============================================================
# 七、补回：你现在在用、但这套 CI 默认没编的
# ==============================================================
# bash（不少脚本 shebang 是 #!/bin/bash，busybox ash 会报错）
CONFIG_PACKAGE_bash=y
# jq（命令行处理 json，排障常用）
CONFIG_PACKAGE_jq=y
# 网页终端，你自己机器上就有
CONFIG_PACKAGE_ttyd=y
CONFIG_PACKAGE_luci-app-ttyd=y

# 网络唤醒（WoL）
# 注意：Config/GENERAL.txt 里有一行 CONFIG_PACKAGE_luci-app-wol=n 主动关掉了它，
# 这里后写覆盖回 y。etherwake 是实际发包的工具，必须一起带上。
CONFIG_PACKAGE_luci-app-wol=y
CONFIG_PACKAGE_etherwake=y

# KMS 激活服务器（vlmcsd）
# 注意：Config/AIROHA-ONU.txt 第 29 行有 CONFIG_PACKAGE_vlmcsd=n 主动关掉了它，这里覆盖回 y。
# 启动后默认监听 TCP 1688，记得在防火墙放行，否则局域网里的 Windows/Office 连不上。
CONFIG_PACKAGE_vlmcsd=y
CONFIG_PACKAGE_luci-app-vlmcsd=y
# 中文语言包（若源码树里没有这两个包，defconfig 会静默丢掉，不影响编译）
CONFIG_PACKAGE_luci-i18n-wol-zh-cn=y
CONFIG_PACKAGE_luci-i18n-vlmcsd-zh-cn=y

# 流量统计（按 IP/主机统计）
# 依赖：kmod-nf-conntrack-netlink 上游已开=y，libnetfilter-acct 需一起带上
#
# ⚠️ 重要：nlbwmon 靠 conntrack 计数。一旦 NPU/flowtable 硬件加速真正生效，
#    已加速的连接会绕过 conntrack，这部分流量统计不到，数字会明显偏低。
#    这不是配置问题，是"硬件加速"和"逐连接统计"的天然冲突。
#    想看真实总带宽，用 状态→实时流量 或 luci-app-statistics（读网卡计数器，不受影响）。
CONFIG_PACKAGE_luci-app-nlbwmon=y
CONFIG_PACKAGE_nlbwmon=y
CONFIG_PACKAGE_libnetfilter-acct=y
CONFIG_PACKAGE_luci-i18n-nlbwmon-zh-cn=y
